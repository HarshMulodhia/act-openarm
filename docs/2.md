```python
"""
Thin wrapper around an openarmisaaclab task.
Handles: env creation, camera setup, observation collection.
"""
from __future__ import annotations
import torch
import numpy as np
from isaaclab.envs import ManagerBasedRLEnv
import gymnasium as gym


class OpenArmIsaacEnv:
    """
    Wraps an openarmisaaclab task registered in Isaac Lab's gym registry.

    Args:
        task_name: e.g. "Isaac-Reach-OpenArm-Unimanual-JointPos-v0"
        num_envs:  number of parallel environments (use 1 for demo collection)
        device:    "cuda:0" or "cpu"
    """

    def __init__(self, task_name: str, num_envs: int = 1, device: str = "cuda:0"):
        self.task_name = task_name
        self.device = device

        # Import OpenArm tasks so they register themselves
        import openarm.tasks  # noqa: F401

        env_cfg = gym.make(task_name).unwrapped.cfg  # read config
        env_cfg.scene.num_envs = num_envs

        self.env: ManagerBasedRLEnv = gym.make(
            task_name,
            cfg=env_cfg,
            render_mode="rgb_array",
        ).unwrapped

        self._setup_cameras()

    def _setup_cameras(self):
        """Add wrist + overhead cameras to the scene."""
        # Isaac Lab camera setup is handled via the task's SceneCfg.
        # For demo collection we rely on the USD sensor variants:
        #   openarmunimanualbase.usd / openarmunimanualsensor.usd
        # No manual camera addition needed if sensor USD is loaded.
        pass

    def reset(self):
        obs, _ = self.env.reset()
        return self._parse_obs(obs)

    def step(self, action: np.ndarray):
        action_tensor = torch.tensor(action, device=self.device, dtype=torch.float32)
        obs, reward, terminated, truncated, info = self.env.step(action_tensor)
        return self._parse_obs(obs), reward.item(), (terminated | truncated).item(), info

    def _parse_obs(self, obs: dict) -> dict:
        """Extract qpos and camera images from Isaac Lab obs dict."""
        return {
            "qpos": obs["policy"].cpu().numpy(),           # joint positions
            # camera images come from info dict when using sensor USDs
        }

    @property
    def action_dim(self) -> int:
        return self.env.action_space.shape[^0]

    def close(self):
        self.env.close()
```

```python
"""
PyTorch Dataset that reads HDF5 episode files produced by 1_collect_demos.py.
Output shape matches what ACT's imitate_episodes.py expects.
"""
from __future__ import annotations
import os
import h5py
import numpy as np
import torch
from torch.utils.data import Dataset


class OpenArmIsaacDataset(Dataset):
    """
    Expects HDF5 files with structure:
        /qpos          [T, n_joints]   float32
        /actions       [T, n_joints]   float32
        /images/cam    [T, H, W, 3]   uint8

    Returns windows of (obs_seq, action_chunk) for ACT training.
    """

    def __init__(
        self,
        root_dir: str,
        camera_key: str = "cam_wrist",
        qpos_key: str = "qpos",
        action_key: str = "actions",
        chunk_size: int = 100,
        image_size: tuple[int, int] = (84, 84),
    ):
        self.chunk_size = chunk_size
        self.image_size = image_size
        self.camera_key = camera_key
        self.qpos_key = qpos_key
        self.action_key = action_key

        self.episode_files = sorted(
            [os.path.join(root_dir, f) for f in os.listdir(root_dir) if f.endswith(".h5")]
        )
        assert len(self.episode_files) > 0, f"No .h5 files found in {root_dir}"

        # Build flat index: (episode_idx, start_t)
        self.index = []
        for ep_idx, fpath in enumerate(self.episode_files):
            with h5py.File(fpath, "r") as f:
                T = f[self.qpos_key].shape[^0]
            for t in range(T - chunk_size):
                self.index.append((ep_idx, t))

    def __len__(self):
        return len(self.index)

    def __getitem__(self, idx: int):
        ep_idx, t = self.index[idx]
        with h5py.File(self.episode_files[ep_idx], "r") as f:
            qpos = torch.tensor(f[self.qpos_key][t], dtype=torch.float32)
            actions = torch.tensor(
                f[self.action_key][t : t + self.chunk_size], dtype=torch.float32
            )
            image = f[f"images/{self.camera_key}"][t]  # [H, W, 3] uint8

        # Resize and normalise image to [3, H, W] float in [0,1]
        import cv2
        image = cv2.resize(image, self.image_size)
        image = torch.tensor(image, dtype=torch.float32).permute(2, 0, 1) / 255.0

        # Pad actions if episode ends before chunk_size
        if actions.shape[^0] < self.chunk_size:
            pad = self.chunk_size - actions.shape[^0]
            actions = torch.cat([actions, actions[-1:].repeat(pad, 1)], dim=0)

        return {
            "image": image,              # [3, H, W]
            "qpos": qpos,                # [n_joints]
            "actions": actions,          # [chunk_size, n_joints]
            "is_pad": torch.zeros(self.chunk_size, dtype=torch.bool),
        }
```

```python
"""
ACT policy wrapper with action-chunk buffer.
Imports directly from the upstream act/ repo cloned at /workspace/act.
"""
from __future__ import annotations
import sys
import os
import numpy as np
import torch

# Add /workspace/act to path so we can import upstream ACT without installing it
ACT_ROOT = os.path.join(os.path.dirname(__file__), "../../../../act")
if ACT_ROOT not in sys.path:
    sys.path.insert(0, ACT_ROOT)

from policy import ACTPolicy  # from /workspace/act/policy.py


class ACTOpenArmPolicy:
    """
    Wraps ACTPolicy with a chunk buffer so the caller can step at 1 Hz
    while ACT reasons at 1/chunk_size Hz.

    Usage:
        policy = ACTOpenArmPolicy.from_checkpoint(ckpt_path, cfg)
        policy.reset()
        for t in range(horizon):
            action = policy.step(image, qpos)
            env.step(action)
    """

    def __init__(self, act_policy: ACTPolicy, chunk_size: int, device: str = "cuda:0"):
        self.policy = act_policy.to(device)
        self.policy.eval()
        self.chunk_size = chunk_size
        self.device = device
        self._chunk_buffer: list[np.ndarray] = []

    @classmethod
    def from_checkpoint(cls, ckpt_path: str, cfg: dict, device: str = "cuda:0"):
        """Load from a .pt checkpoint produced by 2_train_act.py."""
        policy_cfg = {
            "num_queries": cfg["model"]["num_queries"],
            "kl_weight": cfg["train"]["kl_weight"],
            "hidden_dim": cfg["model"]["hidden_dim"],
            "dim_feedforward": cfg["model"]["dim_feedforward"],
            "lr_backbone": 1e-5,
            "backbone": "resnet18",
            "enc_layers": cfg["model"]["enc_layers"],
            "dec_layers": cfg["model"]["dec_layers"],
            "nheads": cfg["model"]["nheads"],
            "camera_names": [cfg["data"]["camera_key"]],
        }
        policy = ACTPolicy(policy_cfg)
        state_dict = torch.load(ckpt_path, map_location=device)
        policy.load_state_dict(state_dict)
        return cls(policy, cfg["data"]["chunk_size"], device)

    def reset(self):
        """Call at the start of each episode."""
        self._chunk_buffer = []

    def step(self, image: np.ndarray, qpos: np.ndarray) -> np.ndarray:
        """
        Args:
            image: [H, W, 3] uint8
            qpos:  [n_joints] float32
        Returns:
            action: [n_joints] float32
        """
        if len(self._chunk_buffer) == 0:
            self._chunk_buffer = self._infer_chunk(image, qpos)

        return self._chunk_buffer.pop(0)

    def _infer_chunk(self, image: np.ndarray, qpos: np.ndarray) -> list[np.ndarray]:
        import torchvision.transforms.functional as TF
        import cv2

        img = cv2.resize(image, (84, 84))
        img_t = torch.tensor(img, dtype=torch.float32).permute(2, 0, 1) / 255.0
        img_t = img_t.unsqueeze(0).unsqueeze(0).to(self.device)  # [1,1,3,H,W]

        qpos_t = torch.tensor(qpos, dtype=torch.float32).unsqueeze(0).to(self.device)

        with torch.no_grad():
            actions = self.policy(qpos_t, img_t)  # [1, chunk_size, n_joints]

        actions_np = actions[^0].cpu().numpy()  # [chunk_size, n_joints]
        return [actions_np[i] for i in range(self.chunk_size)]
```

```python
"""Read / write HDF5 episode files."""
from __future__ import annotations
import h5py
import numpy as np
import os


def save_episode(
    path: str,
    qpos: np.ndarray,        # [T, n_joints]
    actions: np.ndarray,     # [T, n_joints]
    images: dict[str, np.ndarray],  # {"cam_wrist": [T,H,W,3], ...}
    success: bool = False,
):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with h5py.File(path, "w") as f:
        f.create_dataset("qpos", data=qpos, compression="lzf")
        f.create_dataset("actions", data=actions, compression="lzf")
        f.attrs["success"] = success
        grp = f.create_group("images")
        for cam_name, frames in images.items():
            grp.create_dataset(cam_name, data=frames, compression="lzf")


def load_episode(path: str) -> dict:
    with h5py.File(path, "r") as f:
        return {
            "qpos": f["qpos"][:],
            "actions": f["actions"][:],
            "images": {k: f[f"images/{k}"][:] for k in f["images"]},
            "success": bool(f.attrs.get("success", False)),
        }
```

```python
def scripted_reach_policy(obs: dict, goal_qpos: np.ndarray) -> np.ndarray:
    """PD controller toward a fixed goal joint configuration."""

    return np.clip(error * 5.0, -1.0, 1.0)

        t = 0
```

```python
"""
Step 2: Train ACT on collected HDF5 demos.
Delegates to /workspace/act/imitate_episodes.py but uses our dataset adapter.

Usage:
    python 2_train_act.py --config ../configs/act_unimanual_reach.yaml
"""
import argparse
import os
import sys
import yaml
import torch
from torch.utils.data import DataLoader, random_split

sys.path.insert(0, os.path.join(os.path.dirname(__file__), "../src"))
sys.path.insert(0, os.path.join(os.path.dirname(__file__), "../../../act"))

from openarmact.datasets.openarm_isaac_dataset import OpenArmIsaacDataset
from policy import ACTPolicy  # from /workspace/act/policy.py


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--config", required=True)
    args = parser.parse_args()

    cfg = yaml.safe_load(open(args.config))
    device = cfg["task"]["device"]
    ckpt_dir = cfg["train"]["ckpt_dir"]
    os.makedirs(ckpt_dir, exist_ok=True)

    # ── Dataset ──────────────────────────────────────────────────────────────
    dataset = OpenArmIsaacDataset(
        root_dir=cfg["data"]["root_dir"],
        camera_key=cfg["data"]["camera_key"],
        chunk_size=cfg["data"]["chunk_size"],
        image_size=tuple(cfg["data"]["image_size"]),
    )

    val_size = max(1, int(0.1 * len(dataset)))
    train_ds, val_ds = random_split(dataset, [len(dataset) - val_size, val_size])
    train_loader = DataLoader(train_ds, batch_size=cfg["train"]["batch_size"], shuffle=True, num_workers=4)
    val_loader   = DataLoader(val_ds,   batch_size=cfg["train"]["batch_size"], shuffle=False, num_workers=2)

    # ── Model ─────────────────────────────────────────────────────────────────
    policy_cfg = {
        "num_queries":    cfg["data"]["chunk_size"],
        "kl_weight":      cfg["train"]["kl_weight"],
        "hidden_dim":     cfg["model"]["hidden_dim"],
        "dim_feedforward":cfg["model"]["dim_feedforward"],
        "lr_backbone":    1e-5,
        "backbone":       "resnet18",
        "enc_layers":     cfg["model"]["enc_layers"],
        "dec_layers":     cfg["model"]["dec_layers"],
        "nheads":         cfg["model"]["nheads"],
        "camera_names":   [cfg["data"]["camera_key"]],
    }
    policy = ACTPolicy(policy_cfg).to(device)
    optimizer = torch.optim.Adam(policy.parameters(), lr=cfg["train"]["lr"])

    best_val = float("inf")

    for epoch in range(cfg["train"]["num_epochs"]):
        policy.train()
        train_loss = 0.0
        for batch in train_loader:
            qpos   = batch["qpos"].to(device)
            images = batch["image"].unsqueeze(1).to(device)  # [B,1,3,H,W]
            actions = batch["actions"].to(device)
            is_pad  = batch["is_pad"].to(device)

            loss_dict = policy(qpos, images, actions, is_pad)
            loss = loss_dict["loss"]
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            train_loss += loss.item()

        train_loss /= len(train_loader)

        # ── Validation ───────────────────────────────────────────────────────
        if epoch % cfg["train"]["eval_every"] == 0:
            policy.eval()
            val_loss = 0.0
            with torch.no_grad():
                for batch in val_loader:
                    qpos    = batch["qpos"].to(device)
                    images  = batch["image"].unsqueeze(1).to(device)
                    actions = batch["actions"].to(device)
                    is_pad  = batch["is_pad"].to(device)
                    val_loss += policy(qpos, images, actions, is_pad)["loss"].item()
            val_loss /= len(val_loader)
            print(f"[{epoch:04d}] train={train_loss:.4f}  val={val_loss:.4f}")

            if val_loss < best_val:
                best_val = val_loss
                torch.save(policy.state_dict(), os.path.join(ckpt_dir, "best.pt"))

        if epoch % cfg["train"]["save_every"] == 0:
            torch.save(policy.state_dict(), os.path.join(ckpt_dir, f"epoch_{epoch:04d}.pt"))

    print(f"Training complete. Best val loss: {best_val:.4f}")


if __name__ == "__main__":
    main()
```

```python
"""
```

### `scripts/4_deploy_act.py`

```python
"""
Step 4: Deploy ACT with Isaac Sim viewer (stream to /viewer in browser).

Usage:
    ./isaaclab.sh -p scripts/4_deploy_act.py \
        --config configs/act_unimanual_reach.yaml \
        --ckpt   checkpoints/unimanual_reach_act/best.pt \
        --enable_cameras
"""
import argparse, os, sys, yaml
import numpy as np
from isaaclab.app import AppLauncher

parser = argparse.ArgumentParser()
parser.add_argument("--config", required=True)
parser.add_argument("--ckpt",   required=True)
AppLauncher.add_app_launcher_args(parser)
args = parser.parse_args()
app_launcher = AppLauncher(args)
sim_app = app_launcher.app

sys.path.insert(0, os.path.join(os.path.dirname(__file__), "../src"))
sys.path.insert(0, os.path.join(os.path.dirname(__file__), "../../../act"))
from openarmact.envs.openarm_isaac_env import OpenArmIsaacEnv
from openarmact.policies.act_openarm_policy import ACTOpenArmPolicy

def main():
    cfg = yaml.safe_load(open(args.config))
    env = OpenArmIsaacEnv(cfg["task"]["name"], num_envs=1, device=cfg["task"]["device"])
    policy = ACTOpenArmPolicy.from_checkpoint(args.ckpt, cfg, device=cfg["task"]["device"])

    print("Deploying. Open <instance-url>/viewer to observe.")
    episode = 0
    while sim_app.is_running():
        obs = env.reset()
        policy.reset()
        done, t = False, 0
        episode += 1
        print(f"Episode {episode}")
        while not done and t < 500 and sim_app.is_running():
            image = np.zeros((84, 84, 3), dtype=np.uint8)  # replace with cam
            action = policy.step(image, obs["qpos"][^0])
            obs, reward, done, _ = env.step(action)
            t += 1

    env.close()
    sim_app.close()

if __name__ == "__main__":
    main()
```

## How the Repos Connect (Data Flow)

```
openarmisaaclab/
  source/openarm/openarm/tasks/
    managerbased/openarmmanipulation/
      unimanual/reach/  ──────────────────► OpenArmIsaacEnv.__init__(task_name)
      unimanual/lift/                         (your env wrapper)
      bimanual/reach/
      assets/openarmunimanual.py ──────────► USD loaded: openarmunimanualsensor.usd
                                               (provides camera sensors)

openarmactproject/scripts/1_collect_demos.py
  ├─ uses OpenArmIsaacEnv           ← above
  └─ saves HDF5 → data/demos_hdf5/

openarmactproject/scripts/2_train_act.py
  ├─ OpenArmIsaacDataset reads HDF5
  └─ imports ACTPolicy from /workspace/act/policy.py
       └─ DETR VAE backbone from /workspace/act/detr/models/detr_vae.py

openarmactproject/checkpoints/  ← best.pt saved here

openarmactproject/scripts/3_eval_act.py / 4_deploy_act.py
  ├─ ACTOpenArmPolicy.from_checkpoint(best.pt)
  └─ runs in OpenArmIsaacEnv (same task as collection)
```

The `openarm/` repo (hardware docs + website) is only needed when you move from sim to real hardware via ROS 2; it doesn't participate in the training pipeline at all.
